# POS MCP Server Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an MCP server that exposes 4 tools (print_text, print_image, print_diagram, print_barcode) to print to a Bisofice 80mm Ethernet thermal printer from any LLM client.

**Architecture:** Monolite leggero — `server.py` (FastMCP tool definitions) → `renderer.py` (Matplotlib/Graphviz/Pillow → 1-bit PIL.Image) → `printer.py` (ESC/POS via TCP or preview as MCP Image). Config from `pos-mcp.json` via dataclass.

**Tech Stack:** Python 3.12+, FastMCP (mcp[cli]), python-escpos, Pillow, matplotlib, graphviz

```python
# src/pos_mcp/config.py
from dataclasses import dataclass
from pathlib import Path
import json


@dataclass
class PrinterConfig:
    ip: str
    port: int = 9100
    width_px: int = 384
    timeout: int = 5
    cut_after_print: bool = True
    dither: bool = True


def load_config(path: Path) -> PrinterConfig:
    if not path.exists():
        raise FileNotFoundError(f"Config not found: {path}")

    with open(path) as f:
        raw = json.load(f)

    printer = raw.get("printer", {})
    defaults = raw.get("defaults", {})

    return PrinterConfig(
        ip=printer["ip"],
        port=printer.get("port", 9100),
        width_px=printer.get("width_px", 384),
        timeout=printer.get("timeout", 5),
        cut_after_print=defaults.get("cut_after_print", True),
        dither=defaults.get("dither", True),
    )
```

```python
# src/pos_mcp/renderer.py
from PIL import Image, ImageDraw, ImageFont
import io

FONT_SIZES = {"small": 16, "normal": 20, "large": 32}


def render_text(
    text: str,
    width_px: int = 384,
    bold: bool = False,
    size: str = "normal",
    align: str = "left",
) -> Image.Image:
    font_size = FONT_SIZES.get(size, 20)
    try:
        path = "/usr/share/fonts/truetype/dejavu/DejaVuSansMono-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSansMono.ttf"
        font = ImageFont.truetype(path, font_size)
    except OSError:
        font = ImageFont.load_default(size=font_size)

    lines = text.split("\n")
    line_height = font_size + 4
    total_height = line_height * len(lines) + 20

    img = Image.new("1", (width_px, total_height), color=1)
    draw = ImageDraw.Draw(img)

    y = 10
    for line in lines:
        bbox = draw.textbbox((0, 0), line, font=font)
        text_width = bbox[2] - bbox[0]

        if align == "center":
            x = (width_px - text_width) // 2
        elif align == "right":
            x = width_px - text_width - 10
        else:
            x = 10

        draw.text((x, y), line, fill=0, font=font)
        y += line_height

    return img
```

```python
import graphviz as gv


def render_dot(dot_code: str, width_px: int = 384) -> Image.Image:
    try:
        source = gv.Source(dot_code)
        png_data = source.pipe(format="png", engine="dot")
    except Exception as e:
        raise ValueError(f"Invalid DOT code: {e}")

    img = Image.open(io.BytesIO(png_data)).convert("RGB")
    ratio = width_px / img.width
    new_height = int(img.height * ratio)
    img = img.resize((width_px, new_height), Image.LANCZOS)
    return img.convert("1", dither=Image.Dither.FLOYDSTEINBERG)
```

```python
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import math
import re

ALLOWED_IMPORTS = {"matplotlib", "matplotlib.pyplot", "numpy", "math"}
IMPORT_PATTERN = re.compile(r"^\s*(?:import|from)\s+([\w.]+)", re.MULTILINE)


def render_matplotlib(code: str, width_px: int = 384) -> Image.Image:
    imports_found = IMPORT_PATTERN.findall(code)
    for mod in imports_found:
        root = mod.split(".")[0]
        if root not in {"matplotlib", "numpy", "math"}:
            raise ValueError(
                f"Import not allowed: {mod}. Only matplotlib, numpy, math are permitted."
            )

    namespace = {
        "plt": plt,
        "np": np,
        "math": math,
        "matplotlib": matplotlib,
        "__builtins__": {
            "range": range, "len": len, "int": int, "float": float,
            "str": str, "list": list, "tuple": tuple, "dict": dict,
            "zip": zip, "enumerate": enumerate, "min": min, "max": max,
            "sum": sum, "abs": abs, "round": round, "sorted": sorted,
            "reversed": reversed, "print": print,
            "True": True, "False": False, "None": None,
        },
    }

    plt.close("all")
    try:
        exec(code, namespace)
    except Exception as e:
        raise ValueError(f"Matplotlib code error: {e}")

    fig = plt.gcf()
    buf = io.BytesIO()
    fig.savefig(buf, format="png", dpi=203, bbox_inches="tight", pad_inches=0.1)
    plt.close("all")
    buf.seek(0)

    img = Image.open(buf).convert("RGB")
    ratio = width_px / img.width
    new_height = int(img.height * ratio)
    img = img.resize((width_px, new_height), Image.LANCZOS)
    return img.convert("1", dither=Image.Dither.FLOYDSTEINBERG)
```

```python
import base64 as b64module


def prepare_image(
    image_base64: str, width_px: int = 384, dither: bool = True
) -> Image.Image:
    try:
        raw = b64module.b64decode(image_base64)
        img = Image.open(io.BytesIO(raw)).convert("RGB")
    except Exception as e:
        raise ValueError(f"Invalid image data: {e}")

    ratio = width_px / img.width
    new_height = int(img.height * ratio)
    img = img.resize((width_px, new_height), Image.LANCZOS)

    if dither:
        return img.convert("1", dither=Image.Dither.FLOYDSTEINBERG)
    else:
        return img.convert("1", dither=Image.Dither.NONE)
```

```python
# src/pos_mcp/printer.py
from PIL import Image
from escpos.printer import Network
import io
import base64

from pos_mcp.config import PrinterConfig


def preview_image(img: Image.Image) -> str:
    buf = io.BytesIO()
    img.save(buf, format="PNG")
    return base64.b64encode(buf.getvalue()).decode()


def send_to_printer(
    config: PrinterConfig,
    image: Image.Image | None = None,
    text: str | None = None,
    text_options: dict | None = None,
    barcode_data: str | None = None,
    barcode_type: str = "qr",
    barcode_label: str = "",
    cut: bool = True,
) -> str:
    p = Network(config.ip, port=config.port, timeout=config.timeout)

    try:
        if text is not None:
            opts = text_options or {}
            size_map = {"small": (1, 1), "normal": (1, 1), "large": (2, 2)}
            w, h = size_map.get(opts.get("size", "normal"), (1, 1))

            p.set(
                align=opts.get("align", "left").upper(),
                bold=opts.get("bold", False),
                width=w,
                height=h,
            )
            p.text(text)

        if image is not None:
            p.image(image)

        if barcode_data is not None:
            if barcode_type == "qr":
                p.qr(barcode_data, size=6)
                if barcode_label:
                    p.set(align="CENTER")
                    p.text(f"{barcode_label}\n")
            else:
                if barcode_label:
                    p.set(align="CENTER")
                    p.text(f"{barcode_label}\n")
                p.barcode(barcode_data, barcode_type.upper(), 64, 2, "", "")

        if cut:
            p.cut()
    finally:
        p.close()

    return "OK"
```

```python
# src/pos_mcp/server.py
from pathlib import Path
from mcp.server.fastmcp import FastMCP
from mcp.types import ImageContent

from pos_mcp.config import load_config, PrinterConfig
from pos_mcp.renderer import render_text, render_dot, render_matplotlib, prepare_image
from pos_mcp.printer import send_to_printer, preview_image

CONFIG_PATH = Path(__file__).parent.parent.parent / "pos-mcp.json"


def _get_config() -> PrinterConfig:
    return load_config(CONFIG_PATH)


mcp = FastMCP(
    "POS Printer",
    instructions=(
        "MCP server for an 80mm thermal printer. "
        "Use print_text for formatted text, print_image for images, "
        "print_diagram for Graphviz DOT or matplotlib code, "
        "print_barcode for QR codes and barcodes. "
        "Set preview=true on any tool to see what will print without using paper."
    ),
)


@mcp.tool()
def print_text(
    content: str,
    bold: bool = False,
    size: str = "normal",
    align: str = "left",
    cut: bool = True,
    preview: bool = False,
) -> str | list:
    """Print formatted text to the thermal printer.

    Args:
        content: Text to print (supports newlines)
        bold: Bold text
        size: Font size - "small", "normal", or "large"
        align: Text alignment - "left", "center", or "right"
        cut: Cut paper after printing
        preview: Return preview image instead of printing
    """
    config = _get_config()

    if preview:
        img = render_text(
            content, width_px=config.width_px, bold=bold, size=size, align=align
        )
        b64 = preview_image(img)
        return [ImageContent(type="image", data=b64, mimeType="image/png")]

    return send_to_printer(
        config,
        text=content + "\n",
        text_options={"bold": bold, "size": size, "align": align},
        cut=cut,
    )


@mcp.tool()
def print_image(
    image_base64: str,
    dither: bool = True,
    cut: bool = True,
    preview: bool = False,
) -> str | list:
    """Print an image to the thermal printer.

    Args:
        image_base64: PNG or JPG image encoded as base64 string
        dither: Apply Floyd-Steinberg dithering for 1-bit conversion
        cut: Cut paper after printing
        preview: Return preview image instead of printing
    """
    config = _get_config()
    img = prepare_image(image_base64, width_px=config.width_px, dither=dither)

    if preview:
        b64 = preview_image(img)
        return [ImageContent(type="image", data=b64, mimeType="image/png")]

    return send_to_printer(config, image=img, cut=cut)


@mcp.tool()
def print_diagram(
    code: str,
    engine: str = "graphviz",
    cut: bool = True,
    preview: bool = False,
) -> str | list:
    """Render and print a diagram from Graphviz DOT or matplotlib Python code.

    Args:
        code: DOT code (for graphviz) or Python code (for matplotlib)
        engine: Rendering engine - "graphviz" or "matplotlib"
        cut: Cut paper after printing
        preview: Return preview image instead of printing
    """
    config = _get_config()

    if engine == "graphviz":
        img = render_dot(code, width_px=config.width_px)
    elif engine == "matplotlib":
        img = render_matplotlib(code, width_px=config.width_px)
    else:
        raise ValueError(f"Unknown engine: {engine}. Use 'graphviz' or 'matplotlib'.")

    if preview:
        b64 = preview_image(img)
        return [ImageContent(type="image", data=b64, mimeType="image/png")]

    return send_to_printer(config, image=img, cut=cut)


@mcp.tool()
def print_barcode(
    data: str,
    type: str = "qr",
    label: str = "",
    cut: bool = True,
    preview: bool = False,
) -> str | list:
    """Print a barcode or QR code to the thermal printer.

    Args:
        data: Barcode data string
        type: Barcode type - "qr", "ean13", "ean8", "code128", "code39"
        label: Optional text label below the barcode
        cut: Cut paper after printing
        preview: Return preview image instead of printing
    """
    config = _get_config()

    if preview:
        if type == "qr":
            import qrcode

            qr = qrcode.make(data)
            img = qr.resize((config.width_px, config.width_px)).convert("1")
        else:
            from barcode import get_barcode_class
            from barcode.writer import ImageWriter

            bc_class = get_barcode_class(type.upper())
            bc = bc_class(data, writer=ImageWriter())
            buf = __import__("io").BytesIO()
            bc.write(buf)
            buf.seek(0)
            from PIL import Image

            img = Image.open(buf).convert("RGB")
            ratio = config.width_px / img.width
            img = img.resize(
                (config.width_px, int(img.height * ratio))
            ).convert("1")

        b64 = preview_image(img)
        return [ImageContent(type="image", data=b64, mimeType="image/png")]

    return send_to_printer(
        config,
        barcode_data=data,
        barcode_type=type,
        barcode_label=label,
        cut=cut,
    )
```
